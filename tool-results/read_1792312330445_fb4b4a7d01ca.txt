   308→function ReferencesTab({ refs, writable, act }: { refs: BootstrapShape["references"]; writable: boolean; act: (p: Record<string, unknown>, m: string) => Promise<void> }) {
   309→  const [importText, setImportText] = React.useState("");
   310→
   311→  const parseAndImport = async () => {
   312→    const text = importText.trim();
   313→    if (!text) return;
   314→    const isBibtex = text.includes("@");
   315→    const parsed: Array<{ title: string; authors: string; year?: string; venue?: string; kind: string }> = [];
   316→    if (isBibtex) {
   317→      const entries = text.split(/@\w+\{/).slice(1);
   318→      for (const e of entries) {
   319→        const getType = text.split(/@(\w+)\{/);
   320→        const title = e.match(/title\s*=\s*[{"'](.+?)[}"']\s*,?\s*\n/i)?.[1] ?? "";
   321→        const author = e.match(/author\s*=\s*[{"'](.+?)[}"']\s*,?\s*\n/i)?.[1] ?? "";
   322→        const year = e.match(/year\s*=\s*[{"']?(\d{4})[{"']?/i)?.[1] ?? "";
   323→        const journal = e.match(/(journal|booktitle)\s*=\s*[{"'](.+?)[}"']/i)?.[2] ?? "";
   324→        if (title) parsed.push({ title, authors: author.replace(/ and /g, "؛ "), year, venue: journal, kind: journal ? "article" : "book" });
   325→      }
   326→    } else {
   327→      // RIS
   328→      const blocks = text.split(/ER\s*-\s*$/m).filter((b) => b.trim());
   329→      for (const b of blocks) {
   330→        const get = (tag: string) => b.match(new RegExp(`${tag}\\s+-\\s+(.+)`, "i"))?.[1]?.trim() ?? "";
   331→        const title = get("TI") || get("T1");
   332→        if (!title) continue;
   333→        const authors = [...b.matchAll(/AU\s+-\s+(.+)/gi)].map((m) => m[1].trim()).join("؛ ");
   334→        parsed.push({ title, authors, year: get("PY") || get("Y1")?.slice(0, 4), venue: get("JO") || get("JF") || get("T2"), kind: get("TY")?.startsWith("JOUR") ? "article" : "book" });
   335→      }
   336→    }
   337→    if (parsed.length === 0) {
   338→      toast({ title: "هیچ مدخلی شناسایی نشد", description: "قالب BibTeX یا RIS معتبر وارد کنید.", variant: "destructive" });
   339→      return;
   340→    }
   341→    for (const p of parsed) {
   342→      await act({ action: "add_reference", ...p, note: "ورود از BibTeX/RIS — مشخصات پیش از استناد بازبینی شود." }, `مدخل «${p.title.slice(0, 30)}…» افزوده شد`);
   343→    }
   344→    setImportText("");
   345→  };
   346→
   347→  return (
   348→    <div className="grid lg:grid-cols-2 gap-5">
   349→      <Card>
   350→        <CardHeader className="pb-2">
   351→          <CardTitle className="text-sm">ورود کتاب‌شناختی از BibTeX یا RIS</CardTitle>
   352→        </CardHeader>
   353→        <CardContent className="space-y-3">
   354→          <Textarea
   355→            value={importText}
   356→            onChange={(e) => setImportText(e.target.value)}
   357→            placeholder={"مثال BibTeX:\n@article{ratti2005,\n  title={Energy consumption and urban texture},\n  author={Ratti, C. and Baker, N.},\n  journal={Energy and Buildings},\n  year={2005}\n}\n\nیا قالب RIS با TI/AU/PY"}
   358→            className="min-h-40 text-xs font-mono ltr-block"
   359→          />
   360→          {writable && <Button size="sm" onClick={parseAndImport}>تجزیه و افزودن مدخل‌ها</Button>}
   361→          <p className="text-[11px] text-muted-foreground leading-5">
   362→            سامانه فقط مدخل‌های شناسایی‌شده را اضافه می‌کند؛ هیچ منبعی جعل نمی‌شود. مشخصات هر مدخل پیش از استناد نهایی باید بازبینی شود.
   363→          </p>
   364→        </CardContent>
   365→      </Card>
   366→
   367→      <Card>
   368→        <CardHeader className="pb-2"><CardTitle className="text-sm">کتابخانه منابع ({fa(refs.length, 0)} مدخل)</CardTitle></CardHeader>
   369→        <CardContent>
   370→          <ScrollArea className="h-96 pe-2">
   371→            <div className="space-y-3">
   372→              {refs.map((r) => (
   373→                <div key={r.id} className="rounded-xl border p-3">
   374→                  <div className="flex items-center gap-2 mb-1">
   375→                    <Badge variant="outline" className="text-[9px]">
   376→                      {r.kind === "article" ? "مقاله" : r.kind === "book" ? "کتاب" : r.kind === "standard" ? "استاندارد" : r.kind === "tool" ? "ابزار" : r.kind}
   377→                    </Badge>

   379→                  </div>

   397→function ResearchMapGraph({ data }: { data: BootstrapShape }) {
   398→  const base = useBaseOption();
   399→  const { theme } = useTheme();
   400→  const isDark = theme === "dark";
   401→
   402→  const nodes: Array<{ id: string; name: string; category: number; symbolSize: number; value?: string }> = [];
   403→  const links: Array<{ source: string; target: string }> = [];
   404→
   405→  // لایه‌ها: پرسش(0) ← متغیر(1) ← داده(2) ← روش(3) ← آزمایش(4) ← نتیجه(5) ← فصل(6)
   406→  const layerNames = ["پرسش پژوهش", "متغیر", "داده", "روش", "آزمایش", "نتیجه", "فصل رساله"];
   407→  data.questions.forEach((q) => nodes.push({ id: `q-${q.id}`, name: q.code, category: 0, symbolSize: 34, value: q.text }));
   408→  ["ارتفاع بنا", "WWR", "کیفیت پوسته", "سطح اشغال", "حیاط مرکزی"].forEach((v, i) => nodes.push({ id: `v-${i}`, name: v, category: 1, symbolSize: 22 }));
   409→  nodes.push({ id: "d-1", name: "داده نمایشی ریخت‌انرژی", category: 2, symbolSize: 28, value: "۵۱ رکورد نمایشی" });
   410→  nodes.push({ id: "m-1", name: "آمار توصیفی و همبستگی", category: 3, symbolSize: 24 });
   411→  nodes.push({ id: "m-2", name: "K-Means", category: 3, symbolSize: 24 });
   412→  nodes.push({ id: "m-3", name: "رگرسیون OLS (تفکیک گروهی)", category: 3, symbolSize: 24 });
   413→  nodes.push({ id: "e-1", name: "اجرای تحلیل آزمایشگاه", category: 4, symbolSize: 26 });
   414→  nodes.push({ id: "r-1", name: "پروفایل تیپ‌ها", category: 5, symbolSize: 24 });
   415→  nodes.push({ id: "r-2", name: "معیارهای آزمون (MAE/RMSE/R²)", category: 5, symbolSize: 24 });
   416→  data.chapters.forEach((c) => nodes.push({ id: `c-${c.id}`, name: `فصل ${c.order}`, category: 6, symbolSize: 28, value: c.title }));
   417→
   418→  data.questions.forEach((q) => {
   419→    for (let i = 0; i < 5; i++) links.push({ source: `q-${q.id}`, target: `v-${i}` });
   420→  });
   421→  for (let i = 0; i < 5; i++) links.push({ source: `v-${i}`, target: "d-1" });
   422→  links.push({ source: "d-1", target: "m-1" }, { source: "d-1", target: "m-2" }, { source: "d-1", target: "m-3" });
   423→  links.push({ source: "m-1", target: "e-1" }, { source: "m-2", target: "e-1" }, { source: "m-3", target: "e-1" });
   424→  links.push({ source: "e-1", target: "r-1" }, { source: "e-1", target: "r-2" });
   425→  data.chapters.forEach((c) => links.push({ source: "r-1", target: `c-${c.id}` }, { source: "r-2", target: `c-${c.id}` }));
   426→
   427→  const option: React.ComponentProps<typeof EChart>["option"] = {
   428→    ...base,
   429→    tooltip: { ...base.tooltip, formatter: (p: { data?: { value?: string; name?: string } }) => p.data?.value || p.data?.name || "" },
   430→    legend: { ...base.legend, data: layerNames, top: 8, textStyle: { ...base.legend.textStyle, fontSize: 11 } },
   431→    series: [
   432→      {
   433→        type: "graph",
   434→        layout: "force",
   435→        roam: true,
   436→        draggable: true,
   437→        categories: layerNames.map((n, i) => ({ name: n, itemStyle: { color: ["#1f4e6e", "#17877d", "#b4693b", "#d9a441", "#7a8c5c", "#8b5d76", "#56B4E9"][i] } })),
   438→        force: { repulsion: 220, edgeLength: 60 },
   439→        label: { show: true, color: isDark ? "#e8e3d7" : "#1b2a38", fontSize: 10, fontFamily: "Vazirmatn" },
   440→        edgeSymbol: ["none", "arrow"],
   441→        edgeSymbolSize: 7,
   442→        lineStyle: { color: isDark ? "#3e5d78" : "#b9c4ce", curveness: 0.12 },
   443→        data: nodes,
   444→        links,
   445→      },
   446→    ],
   447→  };
   448→
   449→  return (
   450→    <Card>
   451→      <CardHeader className="pb-0">
   452→        <CardTitle className="text-sm">نقشه ارتباطات پژوهش — پرسش ← متغیر ← داده ← روش ← آزمایش ← نتیجه ← فصل</CardTitle>
   453→      </CardHeader>
   454→      <CardContent className="p-4">
   455→        <EChart option={option} height={460} />
   456→        <p className="text-[11px] text-muted-foreground mt-2 leading-5">
   457→          گراف تعاملی است (جابه‌جایی و بزرگ‌نمایی ممکن است). هر نتیجه تحلیل باید در این زنجیره قابل ردیابی به داده، روش و نسخه باشد.
   458→        </p>
   459→      </CardContent>
   460→    </Card>
   461→  );
   462→}
